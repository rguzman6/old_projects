struct Vertex //array of vertices
{
	int id; 
	int num_outgoing_edges; 
	AdjacentEdge* adj_list; //linked list to indicate neighbors
}* vertices; 

struct AdjacentEdge //adjacency list (linked list)
{
	Vertex* neighbor; 
	int weight;
	bool not_used; //initialized to true
	AdjacentEdge* next; 
};

struct Edge //for stack --> contains source node and adjacency list
{
	Vertex* vertex; //source node
	AdjacentEdge* edge; //adjacency list
	Edge* next; //bc linked list use this node; can make a different data structure and not use this (like an array)
}
